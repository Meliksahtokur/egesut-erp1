### Task 1: Event Delegation Sistemi

**Files:** Modify: `index.html`, Create: `js/utils/events.js`

**Step 1: events.js oluştur — merkezi delegate listener**

```js
// js/utils/events.js
// Merkezi event delegation
const ACTION_HANDLERS = {};

function registerAction(action, handler) {
  ACTION_HANDLERS[action] = handler;
}

document.addEventListener('click', (e) => {
  const target = e.target.closest('[data-action]');
  if (!target) return;
  const action = target.dataset.action;
  const handler = ACTION_HANDLERS[action];
  if (handler) {
    e.preventDefault();
    handler(target, e);
  }
});

document.addEventListener('input', (e) => {
  const target = e.target.closest('[data-action-input]');
  if (!target) return;
  const action = target.dataset.actionInput;
  const handler = ACTION_HANDLERS[action];
  if (handler) handler(target, e);
});

document.addEventListener('change', (e) => {
  const target = e.target.closest('[data-action-change]');
  if (!target) return;
  const action = target.dataset.actionChange;
  const handler = ACTION_HANDLERS[action];
  if (handler) handler(target, e);
});
```

**Step 2: HTML'de onclick → data-action dönüşümü**

```html
<!-- ESKI -->
<button onclick="openDet('H123')">Detay</button>
<input oninput="filterA(this.value)">

<!-- YENI -->
<button data-action="open-det" data-id="H123">Detay</button>
<input data-action-input="filter-animals">
```

**Step 3: app.js'de handler'ları registerAction ile kaydet**

```js
registerAction('open-det', (el) => openDet(el.dataset.id));
registerAction('filter-animals', (el) => filterA(el.value));
registerAction('open-modal', (el) => openM(el.dataset.modal));
registerAction('close-modal', (el) => closeM(el.dataset.modal));

// ...tum handler'lar
```

**Step 4: index.html'ye events.js yükle**

```html
<script src="js/utils/events.js?v=3"></script>
```

**Step 5: 212 onclick/oninput'i temizle, data-action ile değiştir**

Tüm `onclick=`, `oninput=`, `onchange=`, `onsubmit=` → sil, yerine `data-action`/`data-action-input`/`data-action-change` koy.

### Task 2: Toast Kuyruk Sistemi

**Files:** Modify: `js/utils/helpers.js`

**Step 1: toast fonksiyonunu kuyruklu hale getir**

```js
const toastQueue = [];
let toastActive = false;

function toast(msg, err = false, duration = 3000) {
  toastQueue.push({ msg, err, duration });
  if (!toastActive) processToastQueue();
}

function processToastQueue() {
  if (!toastQueue.length) { toastActive = false; return; }
  toastActive = true;
  const { msg, err, duration } = toastQueue.shift();
  const t = document.createElement('div');
  t.className = `toast ${err ? 'err' : 'ok'}`;
  t.textContent = msg;
  document.body.appendChild(t);
  requestAnimationFrame(() => t.classList.add('show'));
  setTimeout(() => {
    t.classList.remove('show');
    setTimeout(() => { t.remove(); processToastQueue(); }, 300);
  }, duration);
}
```

**Step 2: CSS güncellemesi (toast konumlandırma)**

