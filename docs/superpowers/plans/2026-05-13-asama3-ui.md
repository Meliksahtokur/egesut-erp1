### Task 1: Event Delegation Sistemi

**Files:** Create: `js/utils/events.js`, Modify: `index.html`

**Step 1: events.js oluştur**

```js
// js/utils/events.js
const ACTIONS = {};

function registerAction(action, handler) {
  ACTIONS[action] = handler;
}

function registerActions(map) {
  Object.entries(map).forEach(([k, v]) => ACTIONS[k] = v);
}

document.addEventListener('click', e => {
  const el = e.target.closest('[data-action]');
  if (!el) return;
  const action = el.dataset.action;
  if (ACTIONS[action]) { e.preventDefault(); ACTIONS[action](el, e); }
});

document.addEventListener('input', e => {
  const el = e.target.closest('[data-action]');
  if (!el || !el.dataset.actionEvent || el.dataset.actionEvent !== 'input') return;
  if (ACTIONS[el.dataset.action]) ACTIONS[el.dataset.action](el, e);
});

document.addEventListener('change', e => {
  const el = e.target.closest('[data-action]');
  if (!el || !el.dataset.actionEvent || el.dataset.actionEvent !== 'change') return;
  if (ACTIONS[el.dataset.action]) ACTIONS[el.dataset.action](el, e);
});
```

### Task 2: Toast Kuyruk Sistemi

**Files:** Modify: `js/utils/helpers.js`

**Step 1: toast'u kuyruklu hale getir**

```js
// helpers.js'teki toast fonksiyonunu DEĞİŞTİR:
const _toastQueue = [];
let _toastActive = false;

function toast(msg, err = false, duration = 3000) {
  _toastQueue.push({ msg, err, duration });
  if (!_toastActive) _processToast();
}

function _processToast() {
  if (!_toastQueue.length) { _toastActive = false; return; }
  _toastActive = true;
  const { msg, err, duration } = _toastQueue.shift();
  const t = document.createElement('div');
  t.className = `toast ${err ? 'err' : 'ok'}`;
  t.textContent = msg;
  // Stack: her toast bir öncekinin üstünde
  const existing = document.querySelectorAll('.toast');
  t.style.bottom = `${20 + existing.length * 55}px`;
  document.body.appendChild(t);
  requestAnimationFrame(() => t.classList.add('show'));
  setTimeout(() => {
    t.classList.remove('show');
    setTimeout(() => { t.remove(); _processToast(); }, 300);
  }, duration);
}
```
